cmake_minimum_required(VERSION 3.22)
project(IsyaGoldNative VERSION 4.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(isya_logic src/logic.cpp src/world_match.cpp src/shn_reader.cpp)
target_include_directories(isya_logic PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(MSVC)
  target_compile_options(isya_logic PRIVATE /W4 /EHsc /utf-8)
else()
  target_compile_options(isya_logic PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(isya_logic_test tests/test_logic.cpp)
target_link_libraries(isya_logic_test PRIVATE isya_logic)
add_test(NAME core_logic COMMAND isya_logic_test)
add_executable(isya_shn_test tests/test_shn.cpp)
target_link_libraries(isya_shn_test PRIVATE isya_logic)
add_test(NAME shn_reader COMMAND isya_shn_test)
add_executable(IsyaShnInspect src/shn_cli.cpp)
target_link_libraries(IsyaShnInspect PRIVATE isya_logic)
if(WIN32)
  find_package(nlohmann_json CONFIG REQUIRED)
  find_package(httplib CONFIG REQUIRED)
  find_package(Tesseract CONFIG REQUIRED)
  add_executable(IsyaGoldNative src/main_win.cpp)
  target_link_libraries(IsyaGoldNative PRIVATE isya_logic nlohmann_json::nlohmann_json httplib::httplib Tesseract::libtesseract user32 gdi32 shell32 ws2_32)
  target_compile_definitions(IsyaGoldNative PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(IsyaGoldNative PRIVATE /W4 /EHsc /utf-8)
  endif()
  set_target_properties(IsyaGoldNative PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../app"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../app")
endif()
